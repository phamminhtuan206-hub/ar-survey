      <button id="save-btn" onclick="saveDataDirectly()" class="w-2/3 py-2.5 bg-cyan-600 hover:bg-cyan-500 rounded-lg text-xs font-bold transition">Lưu vào Google Sheet</button>

  <script>
    // *** CHÈN URL WEB APP GOOGLE APPS SCRIPT CỦA BẠN VÀO ĐÂY ***
    const GAS_WEB_APP_URL = "https://script.google.com/macros/s/AKfycbx3w-5j3oWFKABnpw9HINSjWokCkJExXX3fmR8eDTJg7mnPBlvc4TBNsxd936u8PUA7/exec";

    const video = document.getElementById('video-bg');
    const container = document.getElementById('canvas-container');
    let scene, camera, renderer, raycaster, mouse;
    const points = [], pointMarkers = [];
    let frameLine = null, measuredWidth = 0, measuredHeight = 0, measuredArea = 0;

    async function startCamera() {
      try {
        const stream = await navigator.mediaDevices.getUserMedia({
          video: { facingMode: { ideal: "environment" } }, audio: false
        });
        video.srcObject = stream;
        await video.play();
      } catch (err) {
        alert("Lỗi camera: " + err.message);
      }
    }

    function init3D() {
      scene = new THREE.Scene();
      camera = new THREE.PerspectiveCamera(60, window.innerWidth / window.innerHeight, 0.1, 100);
      camera.position.set(0, 0, 5);

      renderer = new THREE.WebGLRenderer({ alpha: true, antialias: true });
      renderer.setSize(window.innerWidth, window.innerHeight);
      renderer.setPixelRatio(window.devicePixelRatio);
      container.appendChild(renderer.domElement);

      raycaster = new THREE.Raycaster();
      mouse = new THREE.Vector2();

      window.addEventListener('resize', onWindowResize);
      container.addEventListener('pointerdown', onPointerDown);
      animate();
    }

    function onPointerDown(event) {
      if (points.length >= 4) return;
      mouse.x = (event.clientX / window.innerWidth) * 2 - 1;
      mouse.y = -(event.clientY / window.innerHeight) * 2 + 1;
      raycaster.setFromCamera(mouse, camera);
      
      const targetZ = 0;
      const distance = (targetZ - raycaster.ray.origin.z) / raycaster.ray.direction.z;
      const pointPosition = raycaster.ray.origin.clone().add(raycaster.ray.direction.clone().multiplyScalar(distance));

      const marker = new THREE.Mesh(
        new THREE.SphereGeometry(0.12, 16, 16),
        new THREE.MeshBasicMaterial({ color: 0xf59e0b })
      );
      marker.position.copy(pointPosition);
      scene.add(marker);

      points.push(pointPosition);
      pointMarkers.push(marker);
      document.getElementById('point-count').innerText = `${points.length} / 4`;

      if (points.length === 4) {
        drawFrame();
        calculateDimensions();
        document.getElementById('instruction').innerText = "Đã khóa 4 góc!";
        document.getElementById('info-panel').classList.remove('hidden');
      }
    }

    function drawFrame() {
      if (frameLine) scene.remove(frameLine);
      const linePoints = [points[0], points[1], points[2], points[3], points[0]];
      const geometry = new THREE.BufferGeometry().setFromPoints(linePoints);
      frameLine = new THREE.Line(geometry, new THREE.LineBasicMaterial({ color: 0x00f0ff, linewidth: 4 }));
      scene.add(frameLine);
    }

    function calculateDimensions() {
      measuredWidth = points[0].distanceTo(points[1]) * 1.5; 
      measuredHeight = points[1].distanceTo(points[2]) * 1.5;
      measuredArea = measuredWidth * measuredHeight;

      document.getElementById('width-val').innerText = `${measuredWidth.toFixed(2)} m`;
      document.getElementById('height-val').innerText = `${measuredHeight.toFixed(2)} m`;
      document.getElementById('area-val').innerText = `${measuredArea.toFixed(2)} m²`;
    }

    window.resetPoints = function() {
      points.length = 0;
      pointMarkers.forEach(m => scene.remove(m));
      pointMarkers.length = 0;
      if (frameLine) scene.remove(frameLine);
      document.getElementById('point-count').innerText = "0 / 4";
      document.getElementById('instruction').innerText = "Chạm màn hình để chấm 4 góc";
      document.getElementById('info-panel').classList.add('hidden');
    };

    window.saveDataDirectly = function() {
      const saveBtn = document.getElementById('save-btn');
      saveBtn.disabled = true;
      saveBtn.innerText = "Đang lưu...";

      const payload = {
        width: measuredWidth.toFixed(2),
        height: measuredHeight.toFixed(2),
        area: measuredArea.toFixed(2),
        points: points.map(p => ({ x: p.x.toFixed(3), y: p.y.toFixed(3), z: p.z.toFixed(3) }))
      };

      fetch(GAS_WEB_APP_URL, {
        method: "POST",
        mode: "no-cors",
        headers: { "Content-Type": "application/json" },
        body: JSON.stringify(payload)
      })
      .then(() => {
        alert("Đã lưu kết quả đo vào Google Sheet thành công!");
        saveBtn.disabled = false;
        saveBtn.innerText = "Lưu vào Google Sheet";
      })
      .catch(err => {
        alert("Lỗi lưu: " + err);
        saveBtn.disabled = false;
        saveBtn.innerText = "Lưu vào Google Sheet";
      });
    };

    function onWindowResize() {
      camera.aspect = window.innerWidth / window.innerHeight;
      camera.updateProjectionMatrix();
      renderer.setSize(window.innerWidth, window.innerHeight);
    }

    function animate() {
      requestAnimationFrame(animate);
      renderer.render(scene, camera);
    }

    startCamera();
    init3D();
  </script>
